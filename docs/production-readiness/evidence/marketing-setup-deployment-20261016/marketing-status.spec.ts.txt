test('live setup error can be retried without a false setup claim',async({page})=>{
 await fixture(page,true,true);
 await page.route('**/api/marketing/setup',route=>route.fulfill({status:503,json:{error:'Unavailable'}}));
 await page.goto('/marketing');const panel=page.getByRole('region',{name:'Email sending setup'});
 await expect(panel.getByRole('alert')).toHaveText('Email setup could not be checked.');
 await expect(panel.getByRole('link',{name:'Email BookingTours'})).toHaveCount(0);
 await page.route('**/api/marketing/setup',route=>route.fulfill({json:{email_service_configured:true,email_feedback_configured:true,operator_status:'eligible',welcome_back_available:false}}));
 await panel.getByRole('button',{name:'Check again'}).click();
 await expect(panel.getByText('Automatic welcome-back emails are not available yet',{exact:true})).toBeVisible();
 await expect(panel.getByText('BookingTours: enable marketing emails for this operator.')).toHaveCount(0);
});

test('missing operator time zone gives a real support action',async({page})=>{
